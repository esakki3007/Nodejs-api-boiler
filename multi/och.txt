
private normalizeCellValue(value: ExcelJS.CellValue): unknown {
  if (value === null || value === undefined) {
    return null;
  }

  if (typeof value === 'object') {
    // Resolve a shared-string reference if ExcelJS exposes one.
    if ('sharedString' in value) {
      const index = (value as { sharedString: number }).sharedString;
      const sharedString = this.sharedStrings?.[index];

      return sharedString === undefined
        ? null
        : this.normalizeCellValue(sharedString as ExcelJS.CellValue);
    }

    // Rich text
    if ('richText' in value && Array.isArray(value.richText)) {
      return value.richText.map((item) => item.text).join('').trim();
    }

    // Hyperlink
    if ('text' in value) {
      return String(value.text).trim();
    }

    // Formula
    if ('result' in value) {
      return value.result == null
        ? null
        : typeof value.result === 'string'
          ? value.result.trim()
          : value.result;
    }
  }

  return typeof value === 'string' ? value.trim() : value;
}
