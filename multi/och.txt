function getSingleChild(
  parent: XmlObject,
  localName: string,
): { name: string; value: unknown } {
  const matches = Object.entries(parent).filter(
    ([key]) => getLocalName(key) === localName,
  );

  if (matches.length !== 1) {
    throw new SoapStructureError(
      `Expected exactly one ${localName} element`,
    );
  }

  const match = matches[0];

  if (!match) {
    throw new SoapStructureError(
      `Missing ${localName} element`,
    );
  }

  const [name, value] = match;

  return { name, value };
}
