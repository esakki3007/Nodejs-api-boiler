
import type { FastifyPluginAsync } from 'fastify';

import {
  validateCheckRequest,
} from '../schemas/check.schema.js';

const checkRoute: FastifyPluginAsync = async (fastify) => {
  fastify.post('/check', async (request, reply) => {
    try {
      // 1. Receive raw XML from the content-type parser.
      if (typeof request.body !== 'string') {
        return reply.code(400).send({
          message: 'Expected a SOAP XML request',
        });
      }

      // 2. Parse and validate XML syntax.
      const parsedXml = fastify.soapParser.parse(
        request.body,
      );

      // 3. Extract the expected SOAP operation.
      const checkPayload =
        fastify.soapEnvelope.extractOperation(
          parsedXml,
          'check',
        );

      // 4. Validate this API's fields.
      if (!validateCheckRequest(checkPayload)) {
        return reply.code(400).send({
          message: 'Invalid check request',
          errors: validateCheckRequest.errors,
        });
      }

      // 5. Next: call your existing adapter,
      // canonical model and screening orchestrator.
      // We will connect these using your actual services.

      return reply.code(501).send({
        message: 'Screening handler not connected yet',
      });
    } catch (error) {
      if (
        error instanceof Error &&
        (
          error.name === 'SoapXmlParseError' ||
          error.name === 'SoapStructureError'
        )
      ) {
        return reply.code(400).send({
          message: error.message,
        });
      }

      request.log.error(
        { err: error },
        'Partner screening request failed',
      );

      return reply.code(500).send({
        message: 'Internal server error',
      });
    }
  });
};

export default checkRoute;
